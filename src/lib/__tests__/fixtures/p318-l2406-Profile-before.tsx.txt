import { useState } from 'react'
import { Link, useLocation, useParams } from 'react-router'

import { getClient } from '@molecule/app-http'
import { useQuery } from '@molecule/app-query-react'
import { useAuth, useTranslation } from '@molecule/app-react'

import type { AppUser } from '../config.js'
import { ArticlePreview } from '../components/ArticlePreview.js'
import { avatarUrl, http, type ArticlesPage, type Profile as ProfilePayload } from '../lib/api.js'

/**
 * A profile: the author's articles, the articles they favorited, and the
 * follow/unfollow control the spec puts beside their avatar.
 */
export function Profile() {
  const { username = '' } = useParams()
  const location = useLocation()
  const { t } = useTranslation()
  const { user, status } = useAuth<AppUser>()

  const favoritesTab = location.pathname.endsWith('/favorites')
  const viewer = user?.username ?? 'anonymous'
  const [following, setFollowing] = useState<boolean | null>(null)

  const profileQuery = useQuery<{ profile: ProfilePayload }>({
    key: ['profile', username, viewer],
    fetch: async (signal) =>
      (await getClient().get<{ profile: ProfilePayload }>(`/profiles/${encodeURIComponent(username)}`, { signal })).data,
  })

  const articlesQuery = useQuery<ArticlesPage>({
    key: ['profile-articles', username, favoritesTab ? 'favorites' : 'author', viewer],
    fetch: async (signal) => {
      const response = await getClient().get<ArticlesPage>('/articles', {
        signal,
        params: favoritesTab
          ? { favorited: username, limit: 20, offset: 0 }
          : { author: username, limit: 20, offset: 0 },
      })
      return response.data
    },
  })

  const profile = profileQuery.data?.profile
  if (profileQuery.status === 'error') {
    return (
      <div className="container page">
        <p role="alert">{t('profile.notFound', undefined, { defaultValue: 'This profile could not be found.' })}</p>
      </div>
    )
  }
  if (!profile) {
    return (
      <div className="container page">
        <p>{t('common.loading', undefined, { defaultValue: 'Loading...' })}</p>
      </div>
    )
  }

  const isSelf = Boolean(user && user.username === profile.username)
  const isFollowing = following ?? profile.following

  const toggleFollow = async () => {
    const desired = !isFollowing
    setFollowing(desired)
    try {
      const path = `/profiles/${encodeURIComponent(profile.username)}/follow`
      const response = desired
        ? await http().post<{ profile: ProfilePayload }>(path)
        : await http().delete<{ profile: ProfilePayload }>(path)
      if (response.data?.profile) setFollowing(response.data.profile.following)
    } catch {
      setFollowing(!desired)
    }
  }

  return (
    <div className="profile-page">
      <div className="user-info">
        <div className="container">
          <div className="row">
            <div className="col-xs-12 col-md-10 offset-md-1">
              <img src={avatarUrl(profile.image)} className="user-img" alt="" />
              <h4>{profile.username}</h4>
              <p>{profile.bio ?? ''}</p>

              {isSelf ? (
                <Link className="btn btn-sm btn-outline-secondary action-btn" to="/settings">
                  <i className="ion-gear-a" />{' '}
                  {t('profile.editSettings', undefined, { defaultValue: 'Edit Profile Settings' })}
                </Link>
              ) : status === 'authenticated' ? (
                <button type="button" className="btn btn-sm btn-outline-secondary action-btn" onClick={toggleFollow}>
                  <i className="ion-plus-round" />{' '}
                  {isFollowing
                    ? t('article.unfollow', undefined, { defaultValue: 'Unfollow' })
                    : t('article.follow', undefined, { defaultValue: 'Follow' })}{' '}
                  {profile.username}
                </button>
              ) : null}
            </div>
          </div>
        </div>
      </div>

      <div className="container">
        <div className="row">
          <div className="col-xs-12 col-md-10 offset-md-1">
            <div className="articles-toggle">
              <ul className="nav nav-pills outline-active">
                <li className="nav-item">
                  <Link
                    className={favoritesTab ? 'nav-link' : 'nav-link active'}
                    to={`/profile/${encodeURIComponent(profile.username)}`}
                  >
                    {t('profile.myArticles', undefined, { defaultValue: 'My Articles' })}
                  </Link>
                </li>
                <li className="nav-item">
                  <Link
                    className={favoritesTab ? 'nav-link active' : 'nav-link'}
                    to={`/profile/${encodeURIComponent(profile.username)}/favorites`}
                  >
                    {t('profile.favorited', undefined, { defaultValue: 'Favorited Articles' })}
                  </Link>
                </li>
              </ul>
            </div>

            {articlesQuery.status === 'loading' || articlesQuery.status === 'idle' ? (
              <div className="article-preview">
                {t('common.loading', undefined, { defaultValue: 'Loading articles...' })}
              </div>
            ) : (articlesQuery.data?.articles ?? []).length === 0 ? (
              <div className="article-preview">
                {t('profile.noArticles', undefined, { defaultValue: 'No articles are here... yet.' })}
              </div>
            ) : (
              (articlesQuery.data?.articles ?? []).map((article) => (
                <ArticlePreview key={article.slug} article={article} />
              ))
            )}
          </div>
        </div>
      </div>
    </div>
  )
}
