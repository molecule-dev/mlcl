import { useAuth } from '@molecule/app-react'
import { useEffect, useState } from 'react'
import { useNavigate } from 'react-router'

import type { ConduitSessionUser } from '../config.js'
import { api, type Article, type Profile } from '../lib/api.js'
import { t } from '../lib/i18n.js'

export function FavoriteButton({
  article,
  className,
}: {
  article: Article
  className?: string
}): JSX.Element {
  const { user } = useAuth<ConduitSessionUser>()
  const navigate = useNavigate()
  const [favorited, setFavorited] = useState(article.favorited)
  const [count, setCount] = useState(article.favoritesCount)
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    setFavorited(article.favorited)
    setCount(article.favoritesCount)
  }, [article.slug, article.favorited, article.favoritesCount])

  const toggle = async (event: React.MouseEvent): Promise<void> => {
    event.preventDefault()
    event.stopPropagation()
    if (!user) {
      navigate('/login')
      return
    }
    if (busy) return
    const next = !favorited
    setBusy(true)
    setFavorited(next)
    setCount((current) => current + (next ? 1 : -1))
    try {
      const result = next ? await api.favorite(article.slug) : await api.unfavorite(article.slug)
      setFavorited(result.article.favorited)
      setCount(result.article.favoritesCount)
    } catch {
      setFavorited(!next)
      setCount((current) => current + (next ? -1 : 1))
    } finally {
      setBusy(false)
    }
  }

  return (
    <button
      className={`btn btn-sm ${favorited ? 'btn-primary' : 'btn-outline-primary'} ${className ?? ''}`}
      data-mol-id={`favorite-${article.slug}`}
      onClick={toggle}
      type="button"
    >
      <i className="ion-heart" />
      &nbsp;
      {favorited
        ? t('conduit.article.unfavorite', undefined, { defaultValue: 'Unfavorite' })
        : t('conduit.article.favorite', undefined, { defaultValue: 'Favorite' })}
      &nbsp;<span className="counter">{count}</span>
    </button>
  )
}

export function FollowButton({
  profile,
  onChange,
}: {
  profile: Profile
  onChange?: (profile: Profile) => void
}): JSX.Element {
  const { user } = useAuth<ConduitSessionUser>()
  const navigate = useNavigate()
  const [following, setFollowing] = useState(profile.following)
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    setFollowing(profile.following)
  }, [profile.username, profile.following])

  const toggle = async (event: React.MouseEvent): Promise<void> => {
    event.preventDefault()
    if (!user) {
      navigate('/login')
      return
    }
    if (busy) return
    const next = !following
    setBusy(true)
    setFollowing(next)
    try {
      const result = next ? await api.follow(profile.username) : await api.unfollow(profile.username)
      setFollowing(result.profile.following)
      onChange?.(result.profile)
    } catch {
      setFollowing(!next)
    } finally {
      setBusy(false)
    }
  }

  return (
    <button
      className={`btn btn-sm ${following ? 'btn-secondary' : 'btn-outline-secondary'}`}
      data-mol-id={`follow-${profile.username}`}
      onClick={toggle}
      type="button"
    >
      {following
        ? t('conduit.profile.unfollow', { username: profile.username }, {
            defaultValue: 'Unfollow {{username}}',
          })
        : t('conduit.profile.follow', { username: profile.username }, {
            defaultValue: 'Follow {{username}}',
          })}
    </button>
  )
}
